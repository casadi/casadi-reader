cmake_minimum_required(VERSION 3.16)
project(casadi_reader VERSION 0.2.1 LANGUAGES NONE)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/c/src")
  enable_language(C)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cpp/src")
  enable_language(CXX)
endif()
include(GNUInstallDirs)
include(CTest)

function(add_reader language directory)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${directory}/src")
    return()
  endif()
  if(language STREQUAL "c")
    set(extension c)
    set(feature c_std_99)
  else()
    set(extension cpp)
    set(feature cxx_std_11)
  endif()
  set(library casadi_reader_${language})
  add_library(${library} SHARED ${directory}/src/reader.${extension})
  target_compile_features(${library} PUBLIC ${feature})
  target_compile_definitions(${library} PRIVATE CASADI_READER_BUILD)
  target_include_directories(${library} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/${directory}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  set_target_properties(${library} PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0 EXPORT_NAME ${language})
  set(cli casadi-reader-${language})
  add_executable(${cli} ${directory}/src/main.${extension})
  target_link_libraries(${cli} PRIVATE ${library})
  if(APPLE)
    set_target_properties(${cli} PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
  elseif(UNIX)
    set_target_properties(${cli} PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  endif()
  install(TARGETS ${library} ${cli} EXPORT ${library}Targets
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT ${language}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT ${language}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT ${language})
  install(DIRECTORY ${directory}/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} COMPONENT ${language})
  install(EXPORT ${library}Targets NAMESPACE casadi_reader::
    FILE ${library}Config.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/${library} COMPONENT ${language})
  if(BUILD_TESTING)
    add_test(NAME ${language}_decode COMMAND ${cli} ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures/arithmetic.casadi)
    if(language STREQUAL "c")
      add_executable(casadi_reader_c_test c/tests/api.c)
      target_link_libraries(casadi_reader_c_test PRIVATE ${library})
      add_test(NAME c_api COMMAND casadi_reader_c_test ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures)
    endif()
  endif()
endfunction()

# Each source distribution contains only its own implementation and compiler requirement.
add_reader(c c)
add_reader(cpp cpp)
